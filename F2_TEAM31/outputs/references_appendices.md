### A.1 Data Preprocessing and Link-Load Derivation ([`src/prepare_abilene.py`](file:///Users/suryavardhan/Documents/Codex/2026-10-23/he/CN%20project/CN_project_download/src/prepare_abilene.py))

* **Role**: Parses raw Abilene telemetry, extracts the first measured field from the 5-values-per-OD layout in `X01.gz`, maps OD demands to physical link loads using the sparse routing matrix $A$, and filters for the 30 internal directed core links (`link_type == 0`).

```python
# Verbatim excerpt from src/prepare_abilene.py (Lines 28-50)
with gzip.open(RAW / "X01.gz", "rt") as f:
    stored = np.loadtxt(f, dtype=np.float64)
if stored.ndim != 2 or stored.shape[1] != 720:
    raise ValueError(f"Expected 720 columns per time step; got {stored.shape}")

# The public file stores five values per OD pair; the measured value is first.
od = stored.reshape(-1, 144, 5)[:, :, 0]
link_names = [r[0] for r in links]
demand_names = [r[0] for r in demands]

routing = np.zeros((54, 144), dtype=np.float64)
for name, demand, li, di, frac in read_rows(RAW / "A", 5):
    li, di = int(li) - 1, int(di) - 1
    if link_names[li] != name or demand_names[di] != demand:
        raise ValueError("Routing indices disagree with metadata")
    routing[li, di] += float(frac)

loads = od @ routing.T
link_types = np.array([int(r[2]) for r in links])
OUT.parent.mkdir(parents=True, exist_ok=True)
np.savez_compressed(
    OUT, od_flows=od, link_loads_all_54=loads,
    link_loads_internal_30=loads[:, link_types == 0],
    link_names=np.array(link_names),
    internal_link_names=np.array(link_names)[link_types == 0],
)
```

### A.2 Neural Model Architectures ([`src/models.py`](file:///Users/suryavardhan/Documents/Codex/2026-10-23/he/CN%20project/CN_project_download/src/models.py))

* **Role**: Defines the temporal baseline ([`GRUBaseline`](file:///Users/suryavardhan/Documents/Codex/2026-10-23/he/CN%20project/CN_project_download/src/models.py#L6-L15)) and the spatio-temporal Graph-GRU ([`GraphGRU`](file:///Users/suryavardhan/Documents/Codex/2026-10-23/he/CN%20project/CN_project_download/src/models.py#L17-L32)).

```python
# Verbatim complete listing from src/models.py (Lines 6-32)
class GRUBaseline(nn.Module):
    def __init__(self, n_links, hidden=64):
        super().__init__()
        self.gru = nn.GRU(n_links, hidden, batch_first=True)
        self.head = nn.Linear(hidden, n_links)

    def forward(self, x):
        _, state = self.gru(x)
        return self.head(state[-1])


class GraphGRU(nn.Module):
    def __init__(self, n_links, hidden=32):
        super().__init__()
        self.spatial = nn.Linear(1, hidden)
        self.gru = nn.GRU(hidden, hidden, batch_first=True)
        self.head = nn.Linear(hidden, 1)
        self.n_links = n_links

    def forward(self, x, adjacency):
        # Aggregate neighboring link measurements at each input time step.
        z = torch.einsum("ij,btjf->btif", adjacency, x.unsqueeze(-1))
        z = torch.relu(self.spatial(z))
        b, t, n, h = z.shape
        z = z.permute(0, 2, 1, 3).reshape(b * n, t, h)
        _, state = self.gru(z)
        return self.head(state[-1]).reshape(b, n)
```

### A.3 Graph Construction and Normalization ([`src/tune_alpha.py`](file:///Users/suryavardhan/Documents/Codex/2026-10-23/he/CN%20project/CN_project_download/src/tune_alpha.py))

* **Role**: Builds the physical router endpoint graph and the training-only top-3 Pearson correlation similarity graph, applying symmetric Laplacian normalization.

```python
# Verbatim excerpt from src/tune_alpha.py (Lines 47-72)
def norm_adj(a):
    a = a.astype(np.float32).copy()
    np.fill_diagonal(a, 1)
    inv = np.power(np.maximum(a.sum(axis=1), 1e-8), -0.5)
    return (inv[:, None] * a * inv[None, :]).astype(np.float32)


def build_raw_graphs(names, train_series):
    """Build unnormalized physical and similarity adjacency matrices from training data only."""
    n = len(names)
    physical = np.eye(n, dtype=np.float32)
    ends = [str(name).split(",") for name in names]
    for i, (a, b) in enumerate(ends):
        for j in range(i + 1, n):
            c, d = ends[j]
            if {a, b} & {c, d}:
                physical[i, j] = physical[j, i] = 1

    corr = np.nan_to_num(np.corrcoef(train_series, rowvar=False), nan=0.0)
    similar = np.eye(n, dtype=np.float32)
    for i in range(n):
        top = sorted(((corr[i, j], j) for j in range(n) if j != i and corr[i, j] > 0), reverse=True)[:3]
        for value, j in top:
            similar[i, j] = similar[j, i] = max(similar[i, j], value)

    return physical, similar
```

```python
# Verbatim excerpt from src/evaluate_load_aware_v2.py (Lines 117-127)
def compute_observed_load(x_windows, mean, std):
    last_step_unscaled = x_windows[:, -1, :].numpy() * std + mean
    return last_step_unscaled.sum(axis=1)


def classify_regimes(observed_loads, q33_units, q67_units):
    return {
        "Low":    observed_loads <= q33_units,
        "Medium": (observed_loads > q33_units) & (observed_loads <= q67_units),
        "High":   observed_loads > q67_units,
    }
```

### B.3 Note on Live Packet Captures

> [!IMPORTANT]
> **Packet captures (`.pcap` / `.pcapng`) are NOT applicable to this project.**
> This project investigates **offline macroscopic time-series link-load forecasting** using the public Abilene backbone telemetry repository (`X01.gz`). Telemetry is aggregated into 5-minute link-load snapshots. The project does not intercept live physical network traffic, deploy sniffing agents (such as `tcpdump` or Wireshark), or execute packet-level discrete-event simulations (such as NS-3 or OMNeT++). All experimental evaluations proceed from historical origin-destination traffic matrices and routing matrices.
